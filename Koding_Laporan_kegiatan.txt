<script>
    function generateReport() {
        // 1. Ambil nilai dari input form
        const tanggal = document.getElementById('tanggal').value;
        const lokasi = document.getElementById('lokasi').value;
        const fotoInput = document.getElementById('foto').files[0];

        // Validasi sederhana jika ada data yang kosong
        if (!tanggal || !lokasi || !fotoInput) {
            alert("Harap semua data dan foto diisi terlebih dahulu!");
            return;
        }

        // 2. Format tanggal agar lebih rapi (opsional)
        const opsiTanggal = { year: 'numeric', month: 'long', day: 'numeric' };
        const tanggalDiformat = new Date(tanggal).toLocaleDateString('id-ID', opsiTanggal);

        // 3. Masukkan data ke dalam template laporan
        document.getElementById('view-tanggal').innerText = tanggalDiformat;
        document.getElementById('view-lokasi').innerText = lokasi;

        // 4. Membaca file foto dan memasukkannya ke template
        const reader = new FileReader();
        reader.onload = function(e) {
            // Pasang hasil pembacaan gambar ke tag <img> template
            document.getElementById('view-foto').src = e.target.result;
            
            // Tampilkan template sejenak dan panggil dialog print browser
            document.getElementById('report-template').style.display = 'block';
            window.print();
            document.getElementById('report-template').style.display = 'none';
        }
        
        // Menjalankan proses pembacaan foto
        reader.readAsDataURL(fotoInput);
    }
</script>
